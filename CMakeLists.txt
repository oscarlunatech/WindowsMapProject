cmake_minimum_required(VERSION 3.25)
project(Cartograph LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(CTest)

find_package(GDAL CONFIG REQUIRED)
# Also found inside Cartograph.Core, where it is linked PRIVATE. Repeated here
# (cheaply - the result is cached) purely so PROJ::proj is in scope for the
# managed-build props file generated at the bottom of this file; imported
# targets created by find_package are visible only in the directory that
# called it.
find_package(PROJ CONFIG REQUIRED)

add_subdirectory(Cartograph.Core)

add_executable(cartograph_cli src/main.cpp src/viewer.cpp)
target_link_libraries(cartograph_cli PRIVATE Cartograph::Core d2d1 dwrite user32)
# viewer.cpp calls the explicit wide (W) Win32 APIs throughout; UNICODE/_UNICODE
# make macros like IDC_ARROW resolve to their wide-string form to match.
target_compile_definitions(cartograph_cli PRIVATE UNICODE _UNICODE)

# PROJ needs proj.db at runtime for coordinate transforms; it doesn't get
# copied alongside the DLLs by vcpkg's applocal step like the binaries do.
add_custom_command(TARGET cartograph_cli POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/proj/proj.db"
        "$<TARGET_FILE_DIR:cartograph_cli>/proj.db"
)

# --- The bridge to the managed half of the tree (Phase 12) -------------------
#
# Cartograph.Interop is C++/CLI and Cartograph.Shell (Phase 13) is WPF, so both
# are built by MSBuild rather than CMake: CMake's COMMON_LANGUAGE_RUNTIME is
# only evaluated by the Visual Studio generators and every preset here is
# Ninja, and CMake cannot build a WPF project at all. See the Phase 12
# DECISIONS entry.
#
# Rather than hand-maintain a copy of the native link line inside a .vcxproj -
# where it would drift silently the first time Core gains a dependency - it is
# emitted here, from the one place that already knows it. One props file per
# build directory, so x64-debug and x64-release each describe their own
# configuration (vcpkg's debug import libraries have different names) and the
# .vcxproj picks by pointing at the matching directory.
#
# CartographNativeBinDir is the CLI's output directory rather than anything
# newly assembled: vcpkg's applocal deployment has already populated it with
# GDAL/PROJ/GEOS and their whole transitive DLL closure, plus the proj.db
# copied just above. The managed projects copy their runtime from there instead
# of recomputing that closure.
file(GENERATE
    OUTPUT "${CMAKE_BINARY_DIR}/cartograph-native.props"
    CONTENT [==[<?xml version="1.0" encoding="utf-8"?>
<!-- Generated by CMake. Do not edit; edit CMakeLists.txt instead. -->
<Project xmlns="http://schemas.microsoft.com/developer/msbuild/2003">
  <PropertyGroup>
    <CartographNativeBinDir>$<TARGET_FILE_DIR:cartograph_cli></CartographNativeBinDir>
  </PropertyGroup>
  <ItemDefinitionGroup>
    <ClCompile>
      <AdditionalIncludeDirectories>$<JOIN:$<TARGET_PROPERTY:Cartograph.Core,INTERFACE_INCLUDE_DIRECTORIES>,$<SEMICOLON>>;%(AdditionalIncludeDirectories)</AdditionalIncludeDirectories>
    </ClCompile>
    <Link>
      <AdditionalDependencies>$<TARGET_LINKER_FILE:Cartograph.Core>;$<TARGET_LINKER_FILE:GDAL::GDAL>;$<TARGET_LINKER_FILE:PROJ::proj>;d2d1.lib;windowscodecs.lib;ole32.lib;uuid.lib;%(AdditionalDependencies)</AdditionalDependencies>
    </Link>
  </ItemDefinitionGroup>
</Project>
]==]
)
